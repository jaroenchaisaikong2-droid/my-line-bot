========================================================================
       คู่มือรูปแบบข้อความ LINE BOT (MESSAGE TYPES) ฉบับสมบูรณ์
========================================================================
รวบรวมรูปแบบข้อมูลทั้งหมดที่คุณสามารถส่งผ่าน LINE Messaging API ได้
โดยใช้ร่วมกับสคริปต์ PowerShell (จัดรูปแบบเป็น Hash Table @{...})
*กฎเหล็ก: การส่ง 1 ครั้ง (1 Request) สามารถนำรูปแบบเหล่านี้มาต่อกันได้สูงสุด 5 ชิ้น

------------------------------------------------------------------------
 1. รูปแบบพื้นฐาน (Basic Messages) - ใช้งานง่าย ก๊อปปี้ไปวางได้เลย
------------------------------------------------------------------------

[1. ข้อความธรรมดา (Text)]
@{
    type = "text"
    text = "สวัสดีครับ นี่คือข้อความทดสอบ"
}

[2. รูปภาพ (Image)]
*ต้องเป็นลิงก์ HTTPS และเป็นไฟล์รูปภาพ (jpg, png)
@{
    type = "image"
    originalContentUrl = "https://example.com/original.jpg" # รูปเต็ม
    previewImageUrl = "https://example.com/preview.jpg"   # รูปย่อ (แสดงตอนโหลด)
}

[3. วิดีโอ (Video)]
*ต้องเป็นลิงก์ HTTPS (mp4) และต้องมีรูประบุเป็นหน้าปก
@{
    type = "video"
    originalContentUrl = "https://example.com/video.mp4"
    previewImageUrl = "https://example.com/cover.jpg"
}

[4. ไฟล์เสียง (Audio)]
*ต้องเป็นลิงก์ HTTPS (m4a) และระบุความยาวเสียงเป็นมิลลิวินาที
@{
    type = "audio"
    originalContentUrl = "https://example.com/audio.m4a"
    duration = 60000 # 60000 มิลลิวินาที = 1 นาที
}

[5. พิกัดแผนที่ (Location)]
*ระบุชื่อสถานที่ ที่อยู่ และพิกัดละติจูด/ลองจิจูด
@{
    type = "location"
    title = "สถานที่ประชุม"
    address = "ชั้น 2 ห้อง A"
    latitude = 13.7563
    longitude = 100.5018
}

[6. สติกเกอร์ (Sticker)]
*ใช้สติกเกอร์พื้นฐานของ LINE (ดูรหัสได้ที่เว็บ LINE Developers)
@{
    type = "sticker"
    packageId = "446"
    stickerId = "1988"
}

------------------------------------------------------------------------
 2. รูปแบบขั้นสูง (Advanced Messages) - สำหรับสร้างลูกเล่นหรือเมนู
------------------------------------------------------------------------

[7. ข้อความแบบปุ่มกด (Template Message - Buttons)]
*ส่งข้อความพร้อมปุ่มกดให้สมาชิกในกลุ่มคลิกลิงก์ได้
@{
    type = "template"
    altText = "ข้อความนี้มีปุ่มกด (สำหรับแสดงในแจ้งเตือน)"
    template = @{
        type = "buttons"
        thumbnailImageUrl = "https://example.com/cover.jpg"
        imageAspectRatio = "rectangle"
        imageSize = "cover"
        imageBackgroundColor = "#FFFFFF"
        title = "เมนูหลัก"
        text = "กรุณาเลือกรายการที่ต้องการ"
        actions = @(
            @{
                type = "uri"
                label = "คลิกเพื่อดูเอกสาร"
                uri = "https://www.google.com"
            }
        )
    }
}

[8. ข้อความแบบการ์ดเลื่อน (Template Message - Carousel)]
*ส่งแผ่นป้ายที่สามารถใช้นิ้วปัดเลื่อนซ้ายขวาได้ (ส่งได้สูงสุด 10 ป้ายใน 1 ชุด)
@{
    type = "template"
    altText = "ข้อความแบบการ์ดเลื่อน"
    template = @{
        type = "carousel"
        columns = @(
            @{
                thumbnailImageUrl = "https://example.com/img1.jpg"
                title = "งานที่ 1"
                text = "รายละเอียดงานที่ 1"
                actions = @( @{ type = "uri"; label = "ดูรายละเอียด"; uri = "https://www.google.com" } )
            },
            @{
                thumbnailImageUrl = "https://example.com/img2.jpg"
                title = "งานที่ 2"
                text = "รายละเอียดงานที่ 2"
                actions = @( @{ type = "uri"; label = "ดูรายละเอียด"; uri = "https://www.google.com" } )
            }
        )
    }
}

[9. การ์ดอิสระขั้นสุดยอด (Flex Message)]
*เป็นรูปแบบที่จัดหน้าตาได้อิสระที่สุด สวยงามที่สุด (เหมือนที่แบรนด์ดังๆ ใช้) 
*หมายเหตุ: โค้ดของ Flex Message จะค่อนข้างยาวและซับซ้อน แนะนำให้ใช้เครื่องมือ
 "LINE Flex Message Simulator" ของ LINE ในการออกแบบ แล้วนำไฟล์ JSON มาใช้ในสคริปต์

------------------------------------------------------------------------
 ตัวอย่างการประกอบร่างใน PowerShell (ส่ง 3 ชิ้นในการกด 1 ครั้ง)
------------------------------------------------------------------------
$messageArray = @(
    # ชิ้นที่ 1: แผนที่
    @{
        type = "location"
        title = "จุดนัดพบ"
        address = "หน้าอาคาร"
        latitude = 13.8441
        longitude = 100.5543
    },
    # ชิ้นที่ 2: ข้อความ
    @{
        type = "text"
        text = "แผนที่สำหรับเดินทางวันนี้ครับ เจอกันตามหมุดเลย"
    },
    # ชิ้นที่ 3: สติกเกอร์
    @{
        type = "sticker"
        packageId = "11537"
        stickerId = "52002735"
    }
)
